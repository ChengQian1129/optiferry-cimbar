cmake_minimum_required(VERSION 3.16)
project(OptiFerry VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic -ffp-contract=off)
enable_testing()
add_executable(afl2-golden tests/golden.cpp)
target_include_directories(afl2-golden PRIVATE src)
add_test(NAME upstream-golden COMMAND afl2-golden ${CMAKE_CURRENT_SOURCE_DIR}/../protocol/test-vectors/afl2)
find_package(OpenSSL REQUIRED)
add_library(nayuki third_party/nayuki/qrcodegen.cpp)
target_include_directories(nayuki PUBLIC third_party/nayuki)
add_executable(core-test tests/core.cpp)
target_include_directories(core-test PRIVATE src)
target_link_libraries(core-test PRIVATE OpenSSL::Crypto nayuki)
add_test(NAME bfb1-qr-core COMMAND core-test)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2 SDL2_ttf)
add_executable(qsend src/main.cpp)
target_link_libraries(qsend PRIVATE OpenSSL::Crypto nayuki PkgConfig::SDL2)
install(TARGETS qsend RUNTIME DESTINATION bin)
# Embed locked wire vectors so installed qsend --selftest needs no source tree.
file(GLOB AFL2_VECTORS "${CMAKE_CURRENT_SOURCE_DIR}/../protocol/test-vectors/afl2/*.bin")
set(GOLDEN_HEADER "${CMAKE_CURRENT_BINARY_DIR}/golden_data.hpp")
file(WRITE "${GOLDEN_HEADER}" "#pragma once\nstruct GoldenDatum {const char* name; const char* hex;};\ninline const GoldenDatum goldenData[] = {\n")
foreach(VECTOR IN LISTS AFL2_VECTORS)
    get_filename_component(VECTOR_NAME "${VECTOR}" NAME)
    file(READ "${VECTOR}" VECTOR_HEX HEX)
    file(APPEND "${GOLDEN_HEADER}" "{\"${VECTOR_NAME}\",\"${VECTOR_HEX}\"},\n")
endforeach()
file(APPEND "${GOLDEN_HEADER}" "};\n")
target_include_directories(qsend PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
